# Navbar Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a persistent glass-style navbar to all protected pages with breadcrumb navigation on the left and a Discord avatar dropdown on the right.

**Architecture:** Two new components (`Navbar`, `UserMenu`) are added to `src/app/components/shared/`. `Layout` is updated to render `Navbar` above a flex-1 content area. Breadcrumbs are computed from the current URL path using a static lookup array. The avatar dropdown uses shadcn's `DropdownMenu`.

**Tech Stack:** React 18, TypeScript, React Router v7, Tailwind CSS v4, shadcn/ui (`DropdownMenu`), `useAuth` (AuthContext), `useLocation`, `useNavigate`, Vitest + @testing-library/react

import { Link, useLocation } from "react-router";
import { UserMenu } from "./user-menu";

interface BreadcrumbEntry {
  match: (pathname: string) => boolean;
  segments: string[];
}

const BREADCRUMBS: BreadcrumbEntry[] = [
  { match: (p) => p === "/dashboard",          segments: ["Dashboard"] },
  { match: (p) => p === "/campaigns",          segments: ["Dashboard", "Campaigns"] },
  { match: (p) => p === "/history",            segments: ["Dashboard", "History"] },
  { match: (p) => p === "/settings",           segments: ["Dashboard", "Settings"] },
  { match: (p) => p === "/character/new",      segments: ["Dashboard", "New Character"] },
  { match: (p) => p.startsWith("/adventure/"), segments: ["Dashboard", "Campaign"] },
];

export function Navbar() {
  const { pathname } = useLocation();
  const entry = BREADCRUMBS.find((b) => b.match(pathname)) ?? null;
  const isOnDashboard = pathname === "/dashboard";

  return (
    <nav
      className="h-14 flex items-center justify-between px-6 border-b border-white/5"
      style={{ background: "rgba(255,255,255,0.04)", backdropFilter: "blur(12px)" }}
    >
      {/* Left: icon + breadcrumb */}
      <div className="flex items-center gap-3">
        {/* Icon */}
        <div className="w-7 h-7 rounded-md bg-gradient-to-br from-purple-500 to-pink-500 flex items-center justify-center text-white text-sm select-none">
          ⚔
        </div>

        {/* Breadcrumb */}
        {entry && (
          <div className="flex items-center gap-2 text-sm">
            {entry.segments.length === 1 ? (
              <span className="text-muted-foreground">Dashboard</span>
            ) : (
              <>
                {isOnDashboard ? (
                  <span className="text-muted-foreground">Dashboard</span>
                ) : (
                  <Link to="/dashboard" className="text-violet-400 hover:text-violet-300 transition-colors">
                    Dashboard
                  </Link>
                )}
                <span className="text-muted-foreground/40">/</span>
                <span className="text-foreground/90">{entry.segments[1]}</span>
              </>
            )}
          </div>
        )}
      </div>

      {/* Right: user menu */}
      <UserMenu />
    </nav>
  );
}
```

import { describe, it, expect, beforeEach, vi } from "vitest";
import { render, screen, act } from "@testing-library/react";
import { MemoryRouter } from "react-router";
import { AuthContext } from "../../context/auth-context";

const mockNavigate = vi.fn();
vi.mock("react-router", async (importOriginal) => {
  const actual = await importOriginal<typeof import("react-router")>();
  return { ...actual, useNavigate: () => mockNavigate };
});

function makeAuth(overrides: Partial<{ avatar: string | null; username: string }> = {}) {
  const logout = vi.fn();
  const auth = {
    user: { id: "42", username: "Steph", avatar: null, email: null, ...overrides },
    isAuthenticated: true,
    login: () => {},
    logout,
  };
  return { auth, logout };
}

function renderMenu(auth: ReturnType<typeof makeAuth>["auth"]) {
  return render(
    <AuthContext.Provider value={auth}>
      <MemoryRouter>
        <UserMenu />
      </MemoryRouter>
    </AuthContext.Provider>
  );
}

beforeEach(() => {
  mockNavigate.mockReset();
});

describe("UserMenu", () => {
  it("shows initials fallback when avatar is null", () => {
    const { auth } = makeAuth({ avatar: null, username: "Steph" });
    renderMenu(auth);
    expect(screen.getByText("S")).toBeInTheDocument();
  });

  it("renders an img when avatar is non-null", () => {
    const { auth } = makeAuth({ avatar: "abc123" });
    renderMenu(auth);
    const img = screen.getByRole("img");
    expect(img).toHaveAttribute("src", "https://cdn.discordapp.com/avatars/42/abc123.png");
    expect(img).toHaveAttribute("alt", "Steph");
  });

  it("falls back to initials when avatar image errors", async () => {
    const { auth } = makeAuth({ avatar: "abc123", username: "Steph" });
    renderMenu(auth);
    const img = screen.getByRole("img");
    await act(async () => {
      img.dispatchEvent(new Event("error"));
    });
    expect(screen.queryByRole("img")).not.toBeInTheDocument();
    expect(screen.getByText("S")).toBeInTheDocument();
  });

  it("calls logout and navigates to / when Logout is clicked", async () => {
    const { auth, logout } = makeAuth();
    renderMenu(auth);
    // Open dropdown

      screen.getByRole("button", { name: "Open user menu" }).click();

      screen.getByText("Logout").click();

    const { container } = render(
      <AuthContext.Provider value={auth}>
        <MemoryRouter><UserMenu /></MemoryRouter>
      </AuthContext.Provider>

import { useState } from "react";
import { Link, useNavigate } from "react-router";
import { useAuth } from "@/app/context/auth-context";
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuTrigger,
} from "@/app/components/ui/dropdown-menu";

export function UserMenu() {
  const { user, logout } = useAuth();
  const navigate = useNavigate();
  const [avatarError, setAvatarError] = useState(false);

  if (!user) return null;

  const avatarUrl = user.avatar
    ? `https://cdn.discordapp.com/avatars/${user.id}/${user.avatar}.png`
    : null;

  const showImage = avatarUrl !== null && !avatarError;

  function handleLogout() {
    logout();
    navigate("/", { replace: true });
  }

  const avatarElement = showImage ? (
    <img
      src={avatarUrl}
      alt={user.username}
      className="w-8 h-8 rounded-full"
      onError={() => setAvatarError(true)}
    />
  ) : (
    <div className="w-8 h-8 rounded-full bg-gradient-to-br from-purple-500 to-pink-500 flex items-center justify-center text-white text-sm font-semibold select-none">
      {user.username[0].toUpperCase()}
    </div>
  );

  return (
    <DropdownMenu>
      <DropdownMenuTrigger asChild>
        <button
          aria-label="Open user menu"
          className="rounded-full focus:outline-none focus:ring-2 focus:ring-violet-500 focus:ring-offset-2 focus:ring-offset-background"
        >
          {avatarElement}
        </button>
      </DropdownMenuTrigger>
      <DropdownMenuContent align="end" className="w-40">
        <DropdownMenuItem asChild>
          <Link to="/profile">Profile</Link>
        </DropdownMenuItem>
        <DropdownMenuItem onClick={handleLogout} className="text-destructive focus:text-destructive">
          Logout
        </DropdownMenuItem>
      </DropdownMenuContent>
    </DropdownMenu>
  );
}
```

import { motion } from "motion/react";

export function Layout() {
  return (
    <div className="h-screen w-screen bg-background dark relative overflow-hidden">
      {/* Ambient background effects */}
      <div className="absolute inset-0 bg-gradient-to-br from-purple-900/10 via-background to-emerald-900/10 pointer-events-none" />
      <div className="absolute top-0 left-1/4 w-96 h-96 bg-purple-500/5 rounded-full blur-3xl pointer-events-none" />
      <div className="absolute bottom-0 right-1/4 w-96 h-96 bg-emerald-500/5 rounded-full blur-3xl pointer-events-none" />

      {/* Floating particles */}
      {[...Array(20)].map((_, i) => (
        <motion.div
          key={i}
          className="absolute w-1 h-1 bg-purple-400/30 rounded-full pointer-events-none"
          style={{
            left: `${Math.random() * 100}%`,
            top: `${Math.random() * 100}%`,
          }}
          animate={{
            y: [0, -30, 0],
            opacity: [0.3, 0.6, 0.3],
          }}
          transition={{
            duration: 3 + Math.random() * 2,
            repeat: Infinity,
            delay: Math.random() * 2,
          }}
        />
      ))}

      <div className="relative z-10 h-full flex flex-col">
        <Navbar />
        <main className="flex-1 overflow-hidden">
          <Outlet />
        </main>
      </div>
    </div>
  );
}
